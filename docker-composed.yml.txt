version: '3.8'

services:

  mysql:
    image: mysql:8
    container_name: npci-mysql
    environment:
      MYSQL_ROOT_PASSWORD: root
      MYSQL_DATABASE: npcidb
    ports:
      - "3306:3306"
    networks:
      - npci-network

  redis:
    image: redis:7
    container_name: npci-redis
    ports:
      - "6379:6379"
    networks:
      - npci-network

  discovery-server:
    build: ./discovery-server
    container_name: discovery-server
    ports:
      - "8761:8761"
    networks:
      - npci-network

  auth-server:
    build: ./auth-server
    container_name: auth-server
    depends_on:
      - mysql
      - discovery-server
    environment:
      SPRING_DATASOURCE_URL: jdbc:mysql://mysql:3306/npcidb
      SPRING_DATASOURCE_USERNAME: root
      SPRING_DATASOURCE_PASSWORD: root
    ports:
      - "9000:9000"
    networks:
      - npci-network

  api-gateway:
    build: ./api-gateway
    container_name: api-gateway
    depends_on:
      - discovery-server
    ports:
      - "18080:18080"
    networks:
      - npci-network

networks:
  npci-network:
    driver: bridge
